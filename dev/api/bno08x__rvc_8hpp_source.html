27[[nodiscard]] constexpr uint8_t RvcSum(std::span<const uint8_t> bytes, std::size_t first,
28 std::size_t last) noexcept {
29 uint32_t sum = 0;
30 for (std::size_t i = first; i <= last; ++i) {
31 sum += static_cast<uint32_t>(bytes[i]);
32 }
33 return static_cast<uint8_t>(sum & 0xFFU);
34}

44[[nodiscard]] constexpr RvcSumRule RvcCheck(std::span<const uint8_t> frame) noexcept {
45 const uint8_t want = frame[18];
46 if (RvcSum(frame, 2U, 17U) == want) {
47 return RvcSumRule::WithMiMr;
48 }
49 const auto alt = static_cast<uint8_t>(
50 (static_cast<uint32_t>(RvcSum(frame, 2U, 14U)) + static_cast<uint32_t>(frame[17])) & 0xFFU);
51 return (alt == want) ? RvcSumRule::WithoutMiMr : RvcSumRule::None;

126 }

128 // no header pair left: keep a trailing 0xAA, which may start the next one
129 count_ = 0;

133 }
134 return;

136 static_cast<void>(
137 std::copy(buf_.begin() + static_cast<std::ptrdiff_t>(p), buf_.end(), buf_.begin()));

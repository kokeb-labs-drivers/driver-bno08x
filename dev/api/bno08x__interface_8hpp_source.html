77template <typename Port>
78concept CommonHooks = requires {

86 ExactHook<Port, bool(uint8_t, const uint8_t*, std::size_t) noexcept>::Take(&Port::I2cWrite);
87 ExactHook<Port, bool(uint8_t, uint8_t*, std::size_t) noexcept>::Take(&Port::I2cRead);
88 ExactHook<Port, bool() noexcept>::Take(&Port::IntAsserted);
89};

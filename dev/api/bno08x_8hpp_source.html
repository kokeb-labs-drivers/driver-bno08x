150[[nodiscard]] constexpr bool SameTarget(const Op& a, const Op& b) noexcept {
151 const bool features =
152 (a.kind == OpKind::SetFeature) && (b.kind == OpKind::SetFeature) && (a.sensor == b.sensor);
153 const bool power = (a.kind == OpKind::Executable) && (b.kind == OpKind::Executable);
154 return features || power;

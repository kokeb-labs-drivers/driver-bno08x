228constexpr int32_t MountQ30(int8_t code) noexcept {
229 constexpr int32_t kOne = 1'073'741'824;
230 constexpr int32_t kHalf = 536'870'912;
231 constexpr int32_t kRoot = 759'250'125;
232 const int32_t mag = (code < 0) ? -code : code;
233 int32_t v = 0;
234 if (mag == 1) {
235 v = kOne;
236 } else if (mag == 2) {
237 v = kHalf;
238 } else if (mag == 3) {
239 v = kRoot;
240 } else {
241 // zero
242 }
243 return (code < 0) ? -v : v;
244}

293class CalibrationHelper {
294public:
296 [[nodiscard]] CalStep Step() const noexcept {
297 return step_;
298 }
301 void Advance() noexcept {
302 const uint8_t s = detail::ToUnderlying(step_);
303 if ((step_ == CalStep::WaitAccuracy) && !accurate_) {
304 return;
305 }
306 if (step_ != CalStep::Done) {
307 step_ = static_cast<CalStep>(s + 1U); // cast of a value proven below `Done`
308 }
309 }
312 void Observe(Accuracy a) noexcept {
313 accurate_ = (a == Accuracy::Medium) || (a == Accuracy::High);
314 if ((step_ == CalStep::WaitAccuracy) && accurate_) {
315 step_ = CalStep::Save;
316 }
317 }
319 [[nodiscard]] bool ReadyToSave() const noexcept {
320 return accurate_;
321 }
323 void Reset() noexcept {
324 step_ = CalStep::EnableSensors;
325 accurate_ = false;
326 }
328 [[nodiscard]] static constexpr uint32_t MinRateHz() noexcept {
329 return 50;
330 }
331
332private:
333 CalStep step_{CalStep::EnableSensors};
334 bool accurate_{false};
335};

258[[nodiscard]] constexpr bool ToSensorId(uint8_t raw, SensorId& out) noexcept {
259 const auto v = static_cast<uint32_t>(raw);
260 const bool known = ((v >= 0x01U) && (v <= 0x16U)) || ((v >= 0x18U) && (v <= 0x1CU)) ||
261 ((v >= 0x1EU) && (v <= 0x23U)) || ((v >= 0x28U) && (v <= 0x2EU));
262 if (known) {
263 out = static_cast<SensorId>(raw);
264 }
265 return known;
266}

538 [[nodiscard]] constexpr bool TimesValid() const noexcept {
539 const bool non_zero = (startup_timeout_ms > 0U) && (op_timeout_ms > 0U) &&
540 (frs_write_timeout_ms > 0U) && (wake_timeout_us > 0U) &&
541 (resync_window_ms > 0U) && (poll_interval_ms > 0U);
542 const bool pulses = (reset_pulse_us > 0U) && (uart_bsq_retry_ms > 0U);
543 const bool order = wake_timeout_us < (static_cast<uint32_t>(op_timeout_ms) * 1000U);
544 return non_zero && pulses && order;
545 }
546
551 [[nodiscard]] constexpr bool Valid() const noexcept {
552 const bool enums =
553 (detail::ToUnderlying(variant) <= detail::ToUnderlying(Variant::Bno086)) &&
554 (detail::ToUnderlying(read_policy) <= detail::ToUnderlying(ReadPolicy::Fixed));
555 const bool sizes = (fixed_read_bytes >= 4U) && (fixed_read_bytes <= 1024U);
556 const bool counts =
557 (resync_limit > 0U) && (product_id_entries >= 1U) && (product_id_entries <= 5U);
558 const bool addr = (i2c_address == reg::kI2cAddrLow) || (i2c_address == reg::kI2cAddrHigh);
559 return enums && sizes && TimesValid() && counts && addr;
560 }
